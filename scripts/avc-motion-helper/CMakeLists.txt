cmake_minimum_required(VERSION 3.21)
project(BitScopeAvcMotion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(bitscope-avc-motion main.cpp)
if(MSVC)
    target_compile_options(bitscope-avc-motion PRIVATE /utf-8)
endif()
set(FFMPEG_ROOT "" CACHE PATH "FFmpeg development SDK (include and lib)")
if(FFMPEG_ROOT)
    find_path(AV_INCLUDE libavcodec/avcodec.h PATHS "${FFMPEG_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
    find_library(AVCODEC_LIBRARY NAMES avcodec PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    find_library(AVUTIL_LIBRARY NAMES avutil PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    target_include_directories(bitscope-avc-motion PRIVATE "${AV_INCLUDE}")
    target_link_libraries(bitscope-avc-motion PRIVATE "${AVCODEC_LIBRARY}" "${AVUTIL_LIBRARY}")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil)
    target_link_libraries(bitscope-avc-motion PRIVATE PkgConfig::AV)
endif()
