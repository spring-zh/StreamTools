cmake_minimum_required(VERSION 3.21)
project(BitScopeHevcHelper LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC_COMPILER_PREDEFINES OFF)
set(GITL_SOURCE "" CACHE PATH "Patched GitlHEVCAnalyzer checkout")
set(ANALYZER "${GITL_SOURCE}/appgitlhevcdecoder/HM_18.0")
set(HM "${ANALYZER}/HM/source")
if(NOT EXISTS "${HM}/Lib/TLibCommon/CommonDef.h")
    message(FATAL_ERROR "HM submodule missing: ${HM}")
endif()
# Mirror the pinned upstream hm.pri source lists; decoder only, no Qt GUI.
file(GLOB COMMON CONFIGURE_DEPENDS "${HM}/Lib/TLibCommon/*.cpp" "${HM}/Lib/libmd5/*.cpp")
file(GLOB DECODER CONFIGURE_DEPENDS "${HM}/Lib/TLibDecoder/*.cpp")
file(GLOB UTILITIES CONFIGURE_DEPENDS "${HM}/Lib/Utilities/*.cpp")
file(GLOB APP CONFIGURE_DEPENDS "${HM}/App/TAppDecoder/*.cpp")
add_executable(HM_180 ${COMMON} ${DECODER} ${UTILITIES} ${APP}
    "${ANALYZER}/TLibSysuAnalyzer/TSysuAnalyzerOutput.cpp")
target_include_directories(HM_180 PRIVATE "${ANALYZER}" "${HM}/Lib"
    "${HM}/Lib/TLibCommon" "${HM}/Lib/libmd5" "${HM}/Lib/TLibDecoder"
    "${HM}/Lib/Utilities" "${HM}/App/TAppDecoder")
target_compile_definitions(HM_180 PRIVATE ENABLE_ANAYSIS_OUTPUT=1)
if(MSVC)
    target_compile_definitions(HM_180 PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(HM_180 PRIVATE /utf-8 /wd4996 /Zc:strictStrings-)
    set_property(TARGET HM_180 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    target_compile_options(HM_180 PRIVATE -msse4.1)
endif()
